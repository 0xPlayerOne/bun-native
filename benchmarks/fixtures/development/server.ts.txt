import { readFileSync, writeFileSync, renameSync } from 'node:fs';
import { revision } from './revision.ts';
const token = process.env['BUNLY_FIXTURE_TOKEN'];
const stateFile = process.env['BUNLY_FIXTURE_STATE'];
if (!token || !/^[a-f0-9]{48}$/.test(token) || !stateFile) throw new Error('Missing isolated fixture configuration');
const instance = crypto.randomUUID();
function readState(): number {
  const value = JSON.parse(readFileSync(stateFile!, 'utf8'));
  if (!Number.isSafeInteger(value.counter) || value.counter < 0 || value.counter > 10000) throw new Error('Invalid fixture state');
  return value.counter;
}
const server = Bun.serve({
  hostname: '127.0.0.1', port: 0,
  fetch(request) {
    const url = new URL(request.url);
    if (request.headers.get('x-fixture-token') !== token || (request.headers.has('origin') && request.headers.get('origin') !== url.origin)) return new Response('Denied', { status: 403 });
    if (url.pathname !== '/state' || !['GET', 'POST'].includes(request.method)) return new Response('Not found', { status: 404 });
    let counter = readState();
    if (request.method === 'POST') {
      if (counter >= 10000) return new Response('Bound exceeded', { status: 409 });
      counter++;
      writeFileSync(stateFile! + '.next', JSON.stringify({ counter }), { mode: 0o600 });
      renameSync(stateFile! + '.next', stateFile!);
    }
    return Response.json({ revision, instance, counter });
  },
});
console.log(JSON.stringify({ event: 'ready', revision, instance, origin: server.url.origin, pid: process.pid }));
process.on('SIGTERM', () => { server.stop(true); process.exit(0); });
