import { readFile } from 'node:fs/promises';
import { randomUUID } from 'node:crypto';
export function configuration() {
  const origin = process.env['BUNLY_EDIT_ORIGIN'] ?? '';
  const launch = process.env['BUNLY_EDIT_LAUNCH'] ?? '';
  const hostToken = process.env['BUNLY_EDIT_HOST_TOKEN'] ?? '';
  const rendererToken = process.env['BUNLY_EDIT_RENDERER_TOKEN'] ?? '';
  const statePath = process.env['BUNLY_EDIT_STATE'] ?? '';
  const workbenchOrigin = process.env['BUNLY_EDIT_FRAME_ORIGIN'] ?? null;
  const url = new URL(origin);
  if (url.protocol !== 'http:' || url.hostname !== '127.0.0.1' || url.origin !== origin || !statePath ||
      ![launch, hostToken, rendererToken].every(v => /^[a-f0-9]{48}$/.test(v))) throw new Error('Invalid private desktop fixture invocation');
  if (workbenchOrigin !== null) {
    const frame = new URL(workbenchOrigin);
    if (frame.protocol !== 'http:' || frame.hostname !== '127.0.0.1' || frame.origin !== workbenchOrigin || !frame.port || workbenchOrigin === origin)
      throw new Error('Invalid separate workbench frame origin');
  }
  return { origin, launch, hostToken, rendererToken, statePath, workbenchOrigin, instance: randomUUID() };
}
export async function register(config: ReturnType<typeof configuration>, identity: {
  revision: string; framework: string; frameworkVersion: string; runtimeVersion: string;
  engineVersion: string | null; sandbox: 'verified' | 'unverified';
}) {
  const raw = await readFile(config.statePath, 'utf8');
  if (Buffer.byteLength(raw) > 4096) throw new Error('Oversized fixture state');
  const state = JSON.parse(raw);
  if (Object.keys(state).length !== 2 || !/^[a-f0-9]{48}$/.test(state.token) || !Number.isSafeInteger(state.counter) || state.counter < 0) throw new Error('Invalid fixture state');
  const response = await fetch(config.origin + '/host', { method: 'POST', signal: AbortSignal.timeout(10000),
    headers: { 'content-type': 'application/json', 'x-bunly-probe-token': config.hostToken },
    body: JSON.stringify({ ...identity, launch: config.launch, instance: config.instance,
      state: state.token, counter: state.counter, pid: process.pid }),
  });
  if (response.status !== 202) throw new Error('Host registration rejected: ' + response.status);
  console.log(JSON.stringify({ event: 'phase', phase: 'desktop-active' }));
}
export async function waitForQuit(config: ReturnType<typeof configuration>) {
  const deadline = performance.now() + 30000;
  while (performance.now() < deadline) {
    const response = await fetch(config.origin + '/command', {
      headers: { 'x-bunly-probe-token': config.hostToken }, signal: AbortSignal.timeout(1000),
    });
    if (!response.ok) throw new Error('Fixture command channel closed');
    const value = await response.json();
    if (value.command === 'quit') return;
    if (value.command === 'observe-idle') {
      if (!Number.isSafeInteger(value.milliseconds) || value.milliseconds < 250 || value.milliseconds > 10000) throw new Error('Invalid idle interval');
      // No command polling occurs in the measured idle interval.
      await new Promise(resolve => setTimeout(resolve, 1000));
      console.log(JSON.stringify({event:'phase',phase:'idle'}));
      await new Promise(resolve => setTimeout(resolve, value.milliseconds));
      console.log(JSON.stringify({event:'phase',phase:'idle-finished'}));
      continue;
    }
    if (value.command !== 'continue') throw new Error('Unknown fixture command');
  }
  throw new Error('Desktop fixture deadline');
}
/** Keep useful engine failure classes without leaking tokens, URLs or paths. */
export function reportFailure(error: unknown): void {
  const known = ['ERR_BLOCKED_BY_ADMINISTRATOR', 'ERR_CONNECTION_REFUSED', 'ERR_CERT_AUTHORITY_INVALID',
    'ERR_ABORTED', 'ERR_FILE_NOT_FOUND', 'ERR_NAME_NOT_RESOLVED'];
  const message = error instanceof Error ? error.message : '';
  const code = known.find(value => message.includes(value)) ?? 'APPLICATION_FAILED';
  console.error(JSON.stringify({ event: 'fixture-failure', code }));
}

/** Origin policy applies to will-frame-navigate and redirects; no remote allowlist. */
export function navigationAllowed(config: { origin: string; workbenchOrigin: string | null }, target: unknown, isMainFrame: unknown): boolean {
  if (typeof target !== 'string' || typeof isMainFrame !== 'boolean') return false;
  try {
    const url = new URL(target);
    if (url.username || url.password || url.protocol !== 'http:' || url.hostname !== '127.0.0.1') return false;
    return url.origin === config.origin || (!isMainFrame && config.workbenchOrigin !== null && url.origin === config.workbenchOrigin);
  } catch { return false; }
}
