/** Private installed-app fixture. CONTROL_FILE is fixed by the benchmark build. */
import { app } from 'mirinjs';
import { existsSync, readFileSync, writeFileSync } from 'node:fs';
import { configuration, register, waitForQuit, reportFailure } from './host.ts';
import { revision } from './revision.ts';
const controlFile = __CONTROL_FILE__;
const readControl = () => JSON.parse(readFileSync(controlFile, 'utf8'));
const initial = readControl();
// The relaunch comes through Launch Services, so do not depend on inherited env.
process.env.HOME = initial.home;
process.env.MIRIN_UPDATE_BASE_URL = initial.updateOrigin;
for (const [key, value] of Object.entries(initial.environment)) process.env[key] = String(value);
let watchdog: ReturnType<typeof setTimeout>;
async function event(name: string, data: Record<string, unknown> = {}) {
  const response = await fetch(initial.updateOrigin + '/events', { method: 'POST',
    headers: { 'content-type': 'application/json', 'x-lifecycle-token': initial.token },
    body: JSON.stringify({ name, revision, pid: process.pid, ...data }), signal: AbortSignal.timeout(5000) });
  if (!response.ok) throw new Error('Lifecycle report rejected');
}
app.on('ready', () => {
  watchdog = setTimeout(() => app.quit(), 120000);
  void (async () => {
    const config = configuration();
    if (!existsSync(config.statePath)) writeFileSync(config.statePath, JSON.stringify(initial.state), { mode: 0o600, flag: 'wx' });
    if (app.updater.currentVersion !== revision) throw new Error('Installed version differs from compiled revision');
    await event('started', { runtime: Bun.version, version: app.updater.currentVersion });
    await register(config, { revision, framework: 'Mirin', frameworkVersion: '0.1.0-alpha.26',
      runtimeVersion: Bun.version, engineVersion: null, sandbox: 'unverified' });
    await app.windows.open({ title: 'Bunly isolated installed-app acceptance', width: 1120, height: 900,
      url: config.origin + '/#token=' + config.rendererToken });
    await waitForQuit(config);
    if (readControl().apply === true && revision === '0.0.1') {
      const update = await app.updater.checkForUpdate();
      if (update?.version !== '0.0.2') throw new Error('Expected the independently built second version');
      await event('update-available', { version: update.version });
      let received = 0, total = 0;
      await app.updater.download(progress => { received = progress.received; total = progress.total; });
      await event('download-verified', { received, total });
      await event('apply-requested');
      await app.updater.applyAndRelaunch();
    } else {
      await event('closing');
      app.quit();
    }
  })().catch(async error => { reportFailure(error); await event('failed').catch(() => {}); app.quit(); })
    .finally(() => clearTimeout(watchdog));
});
