import { app, BrowserWindow, ipcMain, session } from 'electron';
import { join } from 'node:path';
import { configuration, register, waitForQuit, reportFailure, navigationAllowed } from './host.ts';
import { revision } from './revision.ts';
async function main() {
  const config = configuration();
  if (process.versions.electron !== '44.9.3') throw new Error('Electron pin mismatch');
  const profile = process.env['BUNLY_EDIT_PROFILE'];
  if (!profile) throw new Error('A private test profile is required');
  app.setPath('userData', profile); app.enableSandbox();
  await app.whenReady();
  session.defaultSession.setPermissionRequestHandler((_contents, _permission, callback) => callback(false));
  const window = new BrowserWindow({ title: 'Bunly desktop edit workbench', width: 800, height: 480, show: true,
    webPreferences: { sandbox: true, contextIsolation: true, nodeIntegration: false, preload: join(__dirname, 'preload.cjs') } });
  window.webContents.setWindowOpenHandler(() => ({ action: 'deny' }));
  // A shared-workbench child frame has its own exact local origin. It must
  // not inherit main-frame navigation authority or authorize arbitrary remotes.
  window.webContents.on('will-frame-navigate', event => {
    if (!navigationAllowed(config, event.url, event.isMainFrame)) event.preventDefault();
  });
  window.webContents.on('will-redirect', event => {
    if (!navigationAllowed(config, event.url, event.isMainFrame)) event.preventDefault();
  });
  // Sandbox attestation is performed outside the page's main world. It is a
  // separate preflight navigation, disclosed in the benchmark overhead record.
  let timer: ReturnType<typeof setTimeout>;
  const sandbox = new Promise<void>((resolve, reject) => {
    timer = setTimeout(() => reject(new Error('No sandbox attestation')), 8000);
    ipcMain.on('bunly-edit:sandbox', (event, value) => {
      if (event.sender !== window.webContents || !event.senderFrame || event.senderFrame.parent ||
          event.senderFrame.url !== config.origin + '/preflight') return;
      if (value !== true) reject(new Error('Renderer sandbox not active')); else resolve();
    });
  });
  void sandbox.catch(() => {});
  try {
    await Promise.all([window.loadURL(config.origin + '/preflight'), sandbox]);
    await register(config, { revision, framework: 'Electron', frameworkVersion: process.versions.electron!,
      runtimeVersion: process.versions.node, engineVersion: process.versions.chrome!, sandbox: 'verified' });
    await window.loadURL(config.origin + '/#token=' + config.rendererToken);
    await waitForQuit(config);
  } finally { clearTimeout(timer!); ipcMain.removeAllListeners('bunly-edit:sandbox'); window.close(); }
}
main().then(() => app.quit()).catch(error => {
  reportFailure(error); app.exit(1);
});
