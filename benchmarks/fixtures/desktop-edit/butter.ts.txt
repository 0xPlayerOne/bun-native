import { createWindow, closeWindow } from 'butterframework';
import { configuration, register, waitForQuit, reportFailure } from './host.ts';
import { revision } from './revision.ts';
// The compiled bootstrap window remains counted; this is not a CEF-equivalent profile.
async function main() {
  const config = configuration();
  await register(config, { revision, framework: 'Butter', frameworkVersion: '1.6.3',
    runtimeVersion: Bun.version, engineVersion: null, sandbox: 'unverified' });
  const window = createWindow({ title: 'Bunly reference workbench', width: 1120, height: 900,
    url: config.origin + '/#token=' + config.rendererToken });
  await waitForQuit(config);
  closeWindow(window);
  closeWindow();
}
main().catch(error => { reportFailure(error); process.exitCode = 1; closeWindow(); });
