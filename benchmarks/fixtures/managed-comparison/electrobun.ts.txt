import { BrowserWindow } from 'electrobun/main';
import { comparisonConfiguration, comparisonRegister, comparisonCommands } from './host.ts';
import { revision } from './revision.ts';
const config = comparisonConfiguration();
let window: BrowserWindow | undefined;
try {
  // The edited imported handler is the source of this response, not controller-provided data.
  await comparisonRegister(config, { revision: revision(), framework: 'electrobun', version: '2.0.3-beta.26', identities: null });
  window = new BrowserWindow({ title: 'Managed edit comparison', frame: { width: 1000, height: 680 },
    renderer: 'cef', sandbox: true, url: config.origin + '/#token=' + config.rendererToken });
  if (window.renderer !== 'cef') throw new Error('Expected CEF; no system-webview fallback');
  await comparisonCommands(config);
} catch { console.error('Comparison application failed'); process.exitCode = 1 }
finally { window?.close() }
